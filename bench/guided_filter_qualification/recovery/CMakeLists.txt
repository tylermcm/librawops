cmake_minimum_required(VERSION 3.20)
project(GuidedFilterRecovery LANGUAGES CXX)
set(RAWENGINE_INSTALL_PREFIX "" CACHE PATH "Installed RawEngine prefix")
set(engine_targets "${RAWENGINE_INSTALL_PREFIX}/lib/cmake/RawEngine/RawEngineTargets.cmake")
if(NOT EXISTS "${engine_targets}")
    message(FATAL_ERROR "Set RAWENGINE_INSTALL_PREFIX to a fresh installed engine prefix.")
endif()
include("${engine_targets}")
add_executable(GuidedFilterRecovery ../Recovery.cpp)
target_link_libraries(GuidedFilterRecovery PRIVATE RawEngine::RawEngine)
target_compile_features(GuidedFilterRecovery PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(GuidedFilterRecovery PRIVATE /fp:strict /W4)
endif()
